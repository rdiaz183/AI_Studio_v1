import { experimental_decide } from 'ai';
import { createGatewayProvider } from '@ai-sdk/gateway';

// Initialize the Vercel AI Gateway provider targeting TypeSafe's Jev model
const gateway = createGatewayProvider({
  baseUrl: 'https://ai-gateway.vercel.sh/v1',
  headers: {
    // Authenticates using your project's Vercel OIDC token
    Authorization: `Bearer ${process.env.VERCEL_OIDC_TOKEN}`,
  },
});

export async function POST(req: Request) {
  const { ticketContext } = await req.json();

  // Jev evaluates parallel questions on a single shared state
  const response = await experimental_decide({
    model: gateway.decisionModel('typesafe-ai/jev'),
    state: ticketContext, 
    questions: {
      urgency: {
        type: 'score',
        description: 'Rate the customer urgency from 0 (low) to 5 (critical).',
      },
      category: {
        type: 'choice',
        options: ['bug_report', 'billing', 'feature_request', 'account_issue'],
        description: 'Categorize the core issue of this support ticket.',
      },
      requiresHuman: {
        type: 'boolean',
        description: 'Does this issue require routing to a human agent immediately?',
      }
    },
  });

  return Response.json(response);
}

TypeSafe's Jev is a hosted "System One" decision model that does not expose its native weights for local deployment. Instead, it is consumed through Vercel AI Gateway.
The open-source code for utilizing Jev through Vercel's framework is managed via the official Vercel Labs repository.

Environment Configuration & Setup

Because Jev runs securely through Vercel's infrastructure, you authenticate utilizing the Vercel CLI locally:
1. Install dependencies:bash
npm install ai @ai-sdk/gateway
Use code with caution.
2. Link to your Vercel Project:bash
vercel login
vercel link
Use code with caution.
3. Pull OIDC Credentials (Creates your local environment file populated with a 12-hour VERCEL_OIDC_TOKEN):bash
vercel env pull
Use code with caution.

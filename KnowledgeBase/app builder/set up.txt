import socket, json, os, subprocess

PROJECT_ROOT = r"C:\Users\rdiaz\Documents\Unreal Projects\Code510_AI_Workspace"
STATE_FILE = os.path.join(PROJECT_ROOT, "Bridge", "state_machine.json")

def sanitize_and_write(file_path, content):
    clean_rel = os.path.normpath(file_path.replace("/", "\\")).lstrip("\\")
    full_path = os.path.abspath(os.path.join(PROJECT_ROOT, clean_rel))
    if not full_path.startswith(os.path.abspath(PROJECT_ROOT)):
        raise ValueError(f"Blocked path traversal: {file_path}")
    os.makedirs(os.path.dirname(full_path), exist_ok=True)
    with open(full_path, "w", encoding="utf-8") as f:
        f.write(content)
    try:
        subprocess.run(["git", "add", full_path], cwd=PROJECT_ROOT, capture_output=True)
        subprocess.run(["git", "commit", "-m", f"feat(ssd-sync): Update {os.path.basename(full_path)}"], cwd=PROJECT_ROOT, capture_output=True)
    except Exception:
        pass

def handle_client(conn):
    try:
        chunks = []
        conn.settimeout(2.0)
        while True:
            try:
                chunk = conn.recv(65536)
                if not chunk: break
                chunks.append(chunk)
                if b"\n" in chunk: break
            except socket.timeout: break
        raw_data = b"".join(chunks).decode("utf-8").strip()
        if not raw_data: return

        rpc = json.loads(raw_data)
        if isinstance(rpc, dict) and rpc.get("action") == "get_state" and os.path.exists(STATE_FILE):
            with open(STATE_FILE, "r", encoding="utf-8") as sf:
                conn.sendall((sf.read() + "\n").encode("utf-8"))
            return

        conn.sendall(b'{"status": "OK"}\n')
    except Exception as e:
        conn.sendall((json.dumps({"status": "ERROR", "error": str(e)}) + "\n").encode("utf-8"))
    finally:
        conn.close()

def main():
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind(("0.0.0.0", 9876))
    server.listen(5)
    while True:
        conn, _ = server.accept()
        handle_client(conn)

if __name__ == "__main__":
    main()
